{{- /* ABOUTME: Emits the theme registry and per-theme sprite manifest as browser globals. */ -}} {{- /* ABOUTME: Hugo can see the filesystem; the browser cannot, so it gets told. */ -}} {{- $themes := .Site.Data.themes -}} {{- $manifest := dict -}} {{- $assetFamilies := dict "tiles" "tiles/day_tiles.webp" "dropdown" "dropdown/drop_down_menu.webp" "notes" "notes/middle.webp" "menu" "menu/sign.webp" "cat" "sprites/cat/sheets/walk_east.png" -}} {{- $themeAssets := dict -}} {{- range $themes -}} {{- $id := .id -}} {{- $names := slice -}} {{- $dir := printf "static/%s/sprites/category" $id -}} {{- if fileExists $dir -}} {{- range readDir $dir -}} {{- if strings.HasSuffix .Name ".png" -}} {{- $names = $names | append (strings.TrimSuffix ".png" .Name) -}} {{- end -}} {{- end -}} {{- end -}} {{- $manifest = merge $manifest (dict $id $names) -}} {{- $families := dict -}} {{- range $family, $probe := $assetFamilies -}} {{- $families = merge $families (dict $family (fileExists (printf "static/%s/%s" $id $probe))) -}} {{- end -}} {{- /* Family presence is not enough for the cat: fantasy ships 20 of the 24 direction sheets, so probing one file says "present" while five run directions still 404. List them individually. */ -}} {{- $sheets := slice -}} {{- $catDir := printf "static/%s/sprites/cat/sheets" $id -}} {{- if fileExists $catDir -}} {{- range readDir $catDir -}} {{- if strings.HasSuffix .Name ".png" -}} {{- $sheets = $sheets | append (strings.TrimSuffix ".png" .Name) -}} {{- end -}} {{- end -}} {{- end -}} {{- $families = merge $families (dict "catSheets" $sheets) -}} {{- $themeAssets = merge $themeAssets (dict $id $families) -}} {{- end -}}
